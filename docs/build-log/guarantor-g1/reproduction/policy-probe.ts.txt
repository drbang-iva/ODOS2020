import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { randomUUID } from 'node:crypto';
import { loginForLocalRepair } from '../../scripts/repair-practice-roles.js';
import { createMedplumClient } from '../../mcp/src/fhir-client.js';
import { buildMedplumAccessPolicy, getRoleDeclaration } from '../../mcp/src/authz/roles.js';
import { registerPatientFromDemographics } from '../../mcp/src/clinic/patient-registration-endpoint.js';
import { TEST_FHIR_AUDIT_CONTEXT, TEST_FHIR_AUDIT_RECORDER } from '../../mcp/tests/fhirAuditTestStub.js';
import { createRoleClient } from '../../mcp/tests/liveRoleClient.js';
const runtime=JSON.parse(readFileSync(new URL('./runtime.json',import.meta.url),'utf8'));
assert.equal(runtime.baseUrl,'http://127.0.0.1:19413');
const adminToken=await loginForLocalRepair(runtime);
const headers={Authorization:`Bearer ${adminToken}`,'Content-Type':'application/fhir+json','X-Medplum':'extended'};
const fhir=createMedplumClient({baseUrl:runtime.baseUrl,accessToken:adminToken,audit:TEST_FHIR_AUDIT_RECORDER,auditContext:TEST_FHIR_AUDIT_CONTEXT});
const post=async(resource:any)=>{
  const response=await fetch(`${runtime.baseUrl}/fhir/R4/${resource.resourceType}`,{method:'POST',headers,body:JSON.stringify(resource)});
  const body=await response.json();
  if(resource.resourceType==='AccessPolicy'&&response.status!==201) writeFileSync(new URL('./policy-chain-rejection.json',import.meta.url),JSON.stringify({request:{method:'POST',url:`${runtime.baseUrl}/fhir/R4/AccessPolicy`,headers:{Authorization:'[REDACTED]','Content-Type':'application/fhir+json'},body:resource},response:{status:response.status,body}},null,2)+'\n');
  assert.equal(response.status,201,JSON.stringify(body));
  return body;
};
const runId=randomUUID();
const practitioner=await post({resourceType:'Practitioner',name:[{family:'Synthetic',given:['Staff']}]});
const policy=buildMedplumAccessPolicy(getRoleDeclaration('staff'));
if(process.argv[2]) for(const rule of policy.resource??[]) if(rule.resourceType==='Person'&&rule.interaction?.includes('create')) rule.criteria=process.argv[2];
const staffPolicy=await post(policy);
const input={demographics:{firstName:'Synthetic',middleName:'',lastName:`G1Policy${runId.slice(0,8)}`,preferredName:'',birthDate:'2010-01-02',gender:'female',phones:[{value:'',use:'home'},{value:'',use:'mobile'}],textable:'',email:'',address:'1 Synthetic Way',city:'Greenville',state:'SC',postalCode:'29601'},responsibleParties:[{localId:'guardian',kind:'person',relationship:'legal-guardian',firstName:'Responsible',middleName:'',lastName:'Synthetic',phone:'864-555-0101',address:'1 Synthetic Way',city:'Greenville',state:'SC',postalCode:'29601',financialResponsible:true,consentAuthority:true,primary:true,courtOrderNotes:'',effectiveDate:'2026-08-25',endDate:''}],confirmDuplicate:false};
const registrationActorRole=process.env.G1_REGISTRATION_ACTOR_ROLE??"staff";
assert.ok(["staff","provider","admin"].includes(registrationActorRole));
let submitted:any;
const serviceFhir={...fhir,executeTransactionAsActor:async(...args:any[])=>{submitted=args[0]; return (fhir.executeTransactionAsActor as any)(...args);}};
const registration=await registerPatientFromDemographics(input as never,{staffReference:`Practitioner/${practitioner.id}`,actorRole:registrationActorRole,roles:[registrationActorRole],project:{reference:`Project/${runtime.projectId}`}},{serviceFhir,now:()=> '2026-09-13T12:00:00.000Z'});
assert.equal(registration.status,201);
const patient=(registration.body as any).patient;
const relatedSearch=await fhir.search('RelatedPerson',{patient:`Patient/${patient.id}`});
const related:any=relatedSearch.entry?.[0].resource;
assert.ok(related?.id);
const personSearch=await fhir.search('Person',{relatedperson:`RelatedPerson/${related.id}`});
const person:any=personSearch.entry?.[0].resource;
assert.ok(person?.id);
const client=await createRoleClient({baseUrl:runtime.baseUrl,roleId:'staff',policyReference:`AccessPolicy/${staffPolicy.id}`,patientReference:`Patient/${patient.id}`,practitionerReference:`Practitioner/${practitioner.id}`,projectId:runtime.projectId,runId,adminToken,track:r=>r});
const capture:any[]=[];
const request=async(method:string,path:string,body?:any)=>{
  const requestHeaders={Authorization:`Bearer ${client.token}`,'Content-Type':'application/fhir+json'};
  const response=await fetch(`${runtime.baseUrl}/fhir/R4/${path}`,{method,headers:requestHeaders,...(body?{body:JSON.stringify(body)}:{})});
  const result=await response.json();
  capture.push({request:{method,url:`${runtime.baseUrl}/fhir/R4/${path}`,headers:{...requestHeaders,Authorization:'[REDACTED]'},...(body?{body}:{})},response:{status:response.status,body:result}});
  return response.status;
};
const statuses:any={personRead:await request('GET',`Person/${person.id}`),personUpdate:await request('PUT',`Person/${person.id}`,{...person,name:[{family:'Synthetic',given:['Updated']}]}),relatedUpdate:await request('PUT',`RelatedPerson/${related.id}`,{...related,name:[{family:'Synthetic',given:['Updated']}]}),personCreate:await request('POST','Person',{resourceType:'Person',name:[{family:'Synthetic',given:['New']}],link:[{target:{reference:`RelatedPerson/${related.id}`},assurance:'level2'}]})};
statuses.personDelete = await request('DELETE', `Person/${person.id}`);
assert.equal(person.link?.[0]?.target?.reference, `RelatedPerson/${related.id}`);
const extended=async(type:string,id:string)=>{const response=await fetch(`${runtime.baseUrl}/fhir/R4/${type}/${id}`,{headers}); assert.equal(response.status,200);return response.json();};
const evidence={observedAt:new Date().toISOString(),projectId:runtime.projectId,policy:staffPolicy,registrationRequest:submitted,registrationStatus:registration.status,registrationActorRole,statuses,person:await extended('Person',person.id),relatedPerson:await extended('RelatedPerson',related.id),capture};
writeFileSync(new URL(process.env.G1_POLICY_OUTPUT??(process.argv[2]?'./policy-chain-probe.json':'./policy-probe.json'),import.meta.url),JSON.stringify(evidence,null,2)+'\n');
console.log(JSON.stringify({statuses,personMeta:evidence.person.meta,relatedMeta:evidence.relatedPerson.meta,personRules:staffPolicy.resource.filter((r:any)=>r.resourceType==='Person')},null,2));
