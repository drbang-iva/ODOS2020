from pathlib import Path
import subprocess,json
root=Path.cwd();lib=root/'ui/src/lib/patient-insurance.ts';screen=root/'ui/src/scenes/insurance/PatientInsurance.tsx';out=root/'docs/build-log/insurance-ins2/guards'
original={p:p.read_text() for p in [lib,screen]}
def run(label):
 r=subprocess.run(['node','--import','tsx','--test','tests/patientInsurance.test.tsx'],cwd=root/'ui',capture_output=True,text=True)
 (out/(label+'.log')).write_text(r.stdout+r.stderr)
 return {'exit':r.returncode,'failed':[line for line in r.stdout.splitlines() if line.startswith('not ok')]}
mutations=[
 ('U1',screen,'person.id && !isGuardianRelatedPerson(person)','person.id'),
 ('U2',lib,'input.existingRelatedPerson && !isGuardianRelatedPerson(input.existingRelatedPerson)','input.existingRelatedPerson'),
 ('U3',lib,'active: existing ? existing.active : true','active: draft.active'),
 ('U4',lib,'throw new Error("The subscriber record is not loaded. Reload before saving.");','entries.push(updateEntry({ ...buildRelatedPersonResource(input.draft), id: subscriberReference.split("/")[1] }));'),
 ('U5',lib,'extension.url === CONSENT_AUTHORITY_EXTENSION_URL || extension.url === RESPONSIBLE_PARTY_PRIMARY_EXTENSION_URL','extension.url === CONSENT_AUTHORITY_EXTENSION_URL'),
]
results=[]
try:
 for name,p,before,after in mutations:
  assert original[p].count(before)==1
  p.write_text(original[p].replace(before,after))
  red=run(name+'-red');assert red['exit']!=0 and any('INS2 '+name in f for f in red['failed']),red
  p.write_text(original[p]);green=run(name+'-restored');assert green['exit']==0,green
  results.append({'guard':name,'red':red,'restored':green})
finally:
 for p,s in original.items():p.write_text(s)
(out/'mutations.json').write_text(json.dumps(results,indent=2));print(json.dumps(results,indent=2))
